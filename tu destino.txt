<script>
    // Función para reducir un número a dígito simple o número maestro
    function reduceToMaster(number) {
        if (number === 11 || number === 22 || number === 33) {
            return number;
        }
        while (number > 9) {
            let sum = 0;
            number.toString().split('').forEach(digit => {
                sum += parseInt(digit);
            });
            number = sum;
        }
        return number;
    }

    // Función para reducir una fecha (día, mes o año) respetando maestros
    function reduceDatePart(part) {
        if (part === 11 || part === 22 || part === 33) {
            return part;
        }
        return reduceToMaster(part);
    }

    // Función principal para calcular el número de vida
    function calculateLifePath(day, month, year) {
        // Reducir cada componente por separado (respetando maestros)
        let reducedDay = reduceDatePart(day);
        let reducedMonth = reduceDatePart(month);
        
        // Para el año, reducir a dígito simple (el año no puede ser 11,22,33 directamente)
        let reducedYear = reduceToMaster(year);
        
        // Sumar los tres componentes
        let sum = reducedDay + reducedMonth + reducedYear;
        
        // Reducir el resultado final respetando números maestros
        let finalNumber = reduceToMaster(sum);
        
        return finalNumber;
    }

    // Nombres de cada número
    const numberNames = {
        1: "El Iniciador",
        2: "El Conciliador",
        3: "El Comunicador",
        4: "El Constructor",
        5: "El Aventurero",
        6: "El Cuidador",
        7: "El Buscador",
        8: "El Realizador",
        9: "El Humanitario",
        11: "El Maestro Inspirador",
        22: "El Maestro Constructor",
        33: "El Maestro Sanador"
    };

    // Obtener elementos del DOM
    const dayInput = document.getElementById('day');
    const monthInput = document.getElementById('month');
    const yearInput = document.getElementById('year');
    const resultNumberSpan = document.getElementById('resultNumber');
    const resultNameSpan = document.getElementById('resultName');

    // Función que actualiza el resultado automáticamente
    function updateResult() {
        let day = parseInt(dayInput.value);
        let month = parseInt(monthInput.value);
        let year = parseInt(yearInput.value);
        
        // Validar que los campos sean números válidos
        if (isNaN(day) || isNaN(month) || isNaN(year)) {
            resultNumberSpan.textContent = "—";
            resultNameSpan.textContent = "Introduce tu fecha completa";
            return;
        }
        
        // Validaciones básicas
        if (day < 1 || day > 31) {
            resultNumberSpan.textContent = "?";
            resultNameSpan.textContent = "Día inválido (1-31)";
            return;
        }
        
        if (month < 1 || month > 12) {
            resultNumberSpan.textContent = "?";
            resultNameSpan.textContent = "Mes inválido (1-12)";
            return;
        }
        
        if (year < 1000 || year > 2100) {
            resultNumberSpan.textContent = "?";
            resultNameSpan.textContent = "Año inválido (1000-2100)";
            return;
        }
        
        // Validación de días por mes (básica)
        const daysInMonth = new Date(year, month, 0).getDate();
        if (day > daysInMonth) {
            resultNumberSpan.textContent = "?";
            resultNameSpan.textContent = `El mes ${month} solo tiene ${daysInMonth} días`;
            return;
        }
        
        // Calcular número de vida
        const lifeNumber = calculateLifePath(day, month, year);
        
        // Mostrar resultado
        resultNumberSpan.textContent = lifeNumber;
        resultNameSpan.textContent = numberNames[lifeNumber] || "Destino único";
    }

    // Añadir eventos a los campos
    dayInput.addEventListener('input', updateResult);
    monthInput.addEventListener('input', updateResult);
    yearInput.addEventListener('input', updateResult);

    // Limitar a solo números
    function onlyNumbers(input) {
        input.value = input.value.replace(/[^0-9]/g, '');
    }
    
    dayInput.addEventListener('input', () => onlyNumbers(dayInput));
    monthInput.addEventListener('input', () => onlyNumbers(monthInput));
    yearInput.addEventListener('input', () => onlyNumbers(yearInput));
    
    // Avanzar al siguiente campo automáticamente
    dayInput.addEventListener('input', function() {
        if (this.value.length === 2) monthInput.focus();
    });
    
    monthInput.addEventListener('input', function() {
        if (this.value.length === 2) yearInput.focus();
    });
</script>
